cmake_minimum_required(VERSION 3.22)
project(lexicue_gemma LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(LX_LLAMA_SOURCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
  add_subdirectory(${LX_LLAMA_SOURCE} llama)
  add_library(lexicue_gemma SHARED llama.cpp)
  target_link_libraries(lexicue_gemma PRIVATE llama-common llama)
else()
  add_library(lexicue_gemma SHARED litert.cpp)
  target_include_directories(lexicue_gemma PRIVATE ${LX_LITERT_INCLUDE})
  target_link_libraries(lexicue_gemma PRIVATE ${LX_LITERT_LIBRARY})
endif()
if(APPLE)
  set_target_properties(lexicue_gemma PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(lexicue_gemma PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
if(ANDROID)
  target_link_options(lexicue_gemma PRIVATE "-Wl,-z,max-page-size=16384")
endif()
install(TARGETS lexicue_gemma LIBRARY DESTINATION . RUNTIME DESTINATION .)
